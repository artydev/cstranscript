Add real-time progress steps (queued, transcribing, finalizing).

This document explains how to enhance the Whisper transcription app with live progress indicators using Server-Sent Events (SSE).

1️⃣ Goal

Steps shown to users:

  1. Queued — waiting in line
  2. Transcribing — being processed by Whisper
  3. Finalizing — preparing output
  4. Done — displaying final transcript

2️⃣ Backend Update — WhisperService

Add intermediate SendUpdate() calls:

private async Task ProcessRequestAsync(TranscriptionRequest request, CancellationToken ct)
{
    try
    {
        await SendUpdate(request.Id, "Fichier ajouté à la file d'attente…", false, true, "queued");
        await Task.Delay(500, ct);
        await SendUpdate(request.Id, "Transcription en cours…", false, true, "transcribing");

        var response = await _retryPolicy.ExecuteAsync(async token =>
        {
            using var stream = new FileStream(request.FilePath, FileMode.Open, FileAccess.Read);
            using var content = new MultipartFormDataContent();
            content.Add(new StreamContent(stream), "file", request.Filename);
            content.Add(new StringContent("openai/whisper-large-v3"), "model");
            content.Add(new StringContent("fr"), "language");
            return await _httpClientFactory.CreateClient("Whisper")
                .PostAsync("v1/audio/transcriptions", content, token);
        }, ct);

        await SendUpdate(request.Id, "Finalisation du résultat…", false, true, "finalizing");

        var json = await response.Content.ReadAsStringAsync(ct);
        using var doc = JsonDocument.Parse(json);
        var text = doc.RootElement.GetProperty("text").GetString() ?? string.Empty;

        await SendUpdate(request.Id, text, false, false, "done");
    }
    catch (Exception ex)
    {
        await SendUpdate(request.Id, $"Erreur : {ex.Message}", true, false, "error");
    }
}

Include the new optional parameter in SendUpdate():

private async Task SendUpdate(Guid id, string msg, bool isError, bool isProcessing, string step = "")
{
    using var scope = _serviceProvider.CreateScope();
    var renderer = scope.ServiceProvider.GetRequiredService<HtmlRenderer>();
    var html = await renderer.Dispatcher.InvokeAsync(async () =>
    {
        var parameters = ParameterView.FromDictionary(new()
        {
            { "Message", msg },
            { "IsError", isError },
            { "IsProcessing", isProcessing },
            { "Step", step }
        });
        var result = await renderer.RenderComponentAsync<TranscriptionResult>(parameters);
        return result.ToHtmlString();
    });
    await _statusTracker.NotifyUpdate(id, html);
}

3️⃣ Update TranscriptionResult.razor

Add and use the Step parameter:

@code {
    [Parameter] public string Message { get; set; } = string.Empty;
    [Parameter] public bool IsError { get; set; }
    [Parameter] public bool IsProcessing { get; set; }
    [Parameter] public string Step { get; set; } = string.Empty;
}

@if (IsError)
{
    <div>🔴 @Message</div>
}
else if (IsProcessing)
{
    <div>
        @{
            string icon = Step switch
            {
                "queued" => "⏳",
                "transcribing" => "🎙️",
                "finalizing" => "🧩",
                _ => "🔵"
            };
        }
        <div>@icon @Message</div>
    </div>
}
else
{
    <div>✅ Transcription terminée</div>
    <div id="final-text">@Message</div>
}

4️⃣ JS and CSS Enhancements

document.addEventListener('sse:message', e => {
  const msg = e.detail?.message || '';
  if (msg.includes('queued')) setState('file-state-queued');
  else if (msg.includes('transcribing')) setState('file-state-processing');
  else if (msg.includes('finalizing')) setState('file-state-finalizing');
  else if (msg.includes('terminée')) setState('file-state-ready');
});
.file-state-queued { color: var(--cyan); }
.file-state-processing { color: var(--purple); }
.file-state-finalizing { color: var(--orange); }
.file-state-ready { color: var(--green); }
